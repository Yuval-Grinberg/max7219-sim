cmake_minimum_required(VERSION 3.16)
project(max7219_sim CXX C)

# ---------------------------------------------------------------------------
#  Which sketch to build. Point SKETCH at a sketch folder or at its .ino file:
#     cmake -B build -DSKETCH=C:/Users/me/Documents/Arduino/MyMatrix
# ---------------------------------------------------------------------------
set(SKETCH "${CMAKE_CURRENT_SOURCE_DIR}/examples/Demo" CACHE PATH "Arduino sketch folder or .ino file")
option(SIM_USE_SDL "Open a window (needs SDL2; downloaded automatically if missing)" ON)
option(SIM_DOWNLOAD_SDL "Download and build SDL2 if it isn't installed" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# --- locate the sketch -------------------------------------------------------
get_filename_component(SKETCH "${SKETCH}" ABSOLUTE)
if(IS_DIRECTORY "${SKETCH}")
  set(SKETCH_DIR "${SKETCH}")
  get_filename_component(SKETCH_NAME "${SKETCH}" NAME)
else()
  get_filename_component(SKETCH_DIR "${SKETCH}" DIRECTORY)
  get_filename_component(SKETCH_NAME "${SKETCH}" NAME_WE)
endif()
file(GLOB SKETCH_INOS "${SKETCH_DIR}/*.ino" "${SKETCH_DIR}/*.pde")
list(SORT SKETCH_INOS)
set(SKETCH_MAIN "")
foreach(f IN LISTS SKETCH_INOS)
  get_filename_component(n "${f}" NAME_WE)
  if(n STREQUAL SKETCH_NAME)
    set(SKETCH_MAIN "${f}")
  endif()
endforeach()
if(NOT SKETCH_MAIN)
  list(LENGTH SKETCH_INOS n_inos)
  if(n_inos EQUAL 0)
    message(FATAL_ERROR "No .ino file found in ${SKETCH_DIR}")
  endif()
  list(GET SKETCH_INOS 0 SKETCH_MAIN)
endif()
list(REMOVE_ITEM SKETCH_INOS "${SKETCH_MAIN}")
set(SKETCH_ALL_INOS "${SKETCH_MAIN}" ${SKETCH_INOS})
# extra .cpp/.c files that live next to the sketch (Arduino compiles those too)
file(GLOB SKETCH_EXTRA_SRC "${SKETCH_DIR}/*.cpp" "${SKETCH_DIR}/*.c" "${SKETCH_DIR}/src/*.cpp" "${SKETCH_DIR}/src/*.c")
message(STATUS "Sketch: ${SKETCH_MAIN}")

# --- sketch preprocessor (prototypes, tabs, #line) --------------------------
add_executable(sketchgen tools/sketchgen.cpp)
set(SKETCH_CPP "${CMAKE_CURRENT_BINARY_DIR}/sketch_${SKETCH_NAME}.cpp")
add_custom_command(
  OUTPUT "${SKETCH_CPP}"
  COMMAND sketchgen "${SKETCH_CPP}" "${SKETCH_NAME}" ${SKETCH_ALL_INOS}
  DEPENDS sketchgen ${SKETCH_ALL_INOS}
  COMMENT "Preparing sketch ${SKETCH_NAME}"
  VERBATIM)

# --- Arduino libraries: everything under libraries/ plus the sketch's own libraries/ folder
set(LIB_SEARCH_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/libraries" "${SKETCH_DIR}/libraries")
set(LIB_SOURCES "")
set(LIB_INCLUDES "")
foreach(root IN LISTS LIB_SEARCH_DIRS)
  if(IS_DIRECTORY "${root}")
    file(GLOB libs LIST_DIRECTORIES true "${root}/*")
    foreach(lib IN LISTS libs)
      if(IS_DIRECTORY "${lib}")
        if(IS_DIRECTORY "${lib}/src")
          set(libsrc "${lib}/src")
          file(GLOB_RECURSE srcs "${libsrc}/*.cpp" "${libsrc}/*.c")
        else()
          set(libsrc "${lib}")
          file(GLOB srcs "${lib}/*.cpp" "${lib}/*.c" "${lib}/utility/*.cpp" "${lib}/utility/*.c")
        endif()
        list(FILTER srcs EXCLUDE REGEX "/(examples|extras|test|tests)/")
        list(FILTER srcs EXCLUDE REGEX "glcdfont\\.c$")  # #included by Adafruit_GFX.cpp
        list(APPEND LIB_SOURCES ${srcs})
        list(APPEND LIB_INCLUDES "${libsrc}")
        if(IS_DIRECTORY "${libsrc}/utility")
          list(APPEND LIB_INCLUDES "${libsrc}/utility")
        endif()
      endif()
    endforeach()
  endif()
endforeach()

set(CORE_SOURCES core/Arduino.cpp core/Print.cpp core/WString.cpp core/SPI.cpp)
set(SIM_SOURCES sim/main.cpp sim/emulator.cpp sim/render.cpp sim/frontend_sdl.cpp)

set(EXE_NAME "${SKETCH_NAME}_sim")
string(REGEX REPLACE "[^A-Za-z0-9_]" "_" EXE_NAME "${EXE_NAME}")
add_executable(${EXE_NAME} ${SIM_SOURCES} ${CORE_SOURCES} ${LIB_SOURCES} ${SKETCH_EXTRA_SRC} "${SKETCH_CPP}")
target_include_directories(${EXE_NAME} PRIVATE core ${LIB_INCLUDES} "${SKETCH_DIR}" "${SKETCH_DIR}/src")
target_compile_definitions(${EXE_NAME} PRIVATE _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS NOMINMAX)
find_package(Threads REQUIRED)
target_link_libraries(${EXE_NAME} PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(${EXE_NAME} PRIVATE /W3 /utf-8 /permissive /wd4244 /wd4267 /wd4305 /wd4996)
else()
  target_compile_options(${EXE_NAME} PRIVATE -Wall -Wno-unused-variable -Wno-unused-but-set-variable
                         -Wno-sign-compare -Wno-narrowing -Wno-overloaded-virtual -fpermissive)
  if(MINGW)
    target_link_options(${EXE_NAME} PRIVATE -static)
  endif()
endif()

# --- SDL2 window -------------------------------------------------------------
if(SIM_USE_SDL)
  find_package(SDL2 QUIET CONFIG)
  if(NOT SDL2_FOUND)
    find_package(SDL2 QUIET)
  endif()
  if(SDL2_FOUND)
    message(STATUS "Using installed SDL2")
  elseif(SIM_DOWNLOAD_SDL)
    message(STATUS "SDL2 not installed — downloading it (first build takes a minute or two)")
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.9/SDL2-2.30.9.tar.gz)
    FetchContent_MakeAvailable(SDL2)
    set(SDL2_FOUND TRUE)
  endif()
  if(SDL2_FOUND)
    if(TARGET SDL2::SDL2-static)
      target_link_libraries(${EXE_NAME} PRIVATE SDL2::SDL2-static)
    elseif(TARGET SDL2::SDL2)
      target_link_libraries(${EXE_NAME} PRIVATE SDL2::SDL2)
    else()
      target_include_directories(${EXE_NAME} PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(${EXE_NAME} PRIVATE ${SDL2_LIBRARIES})
    endif()
    target_compile_definitions(${EXE_NAME} PRIVATE SIM_HAVE_SDL)
  else()
    message(WARNING "SDL2 not available: the simulator will draw the matrix in the terminal.")
  endif()
endif()

message(STATUS "Executable: ${EXE_NAME}")
